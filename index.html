/** * Arquivo Principal do Backend (Google Apps Script) * Gerencia a comunicação entre a planilha do Google e a Interface Web. */ function doGet() { return HtmlService.createTemplateFromFile('Index') .evaluate() .setTitle('Gestão de Estudantes - CEJA Dr. Gerardo Camelo Madeira') .addMetaTag('viewport', 'width=device-width, initial-scale=1') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } // 1. CARREGAR E CALCULAR DADOS EM TEMPO REAL function obterDadosSincronizados() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let aba = ss.getSheetByName('Matriculas') || ss.getSheets()[0]; const dados = aba.getDataRange().getValues(); if (dados.length <= 1) return []; // Retorna lista vazia se só houver cabeçalho dados.shift(); // Remove o cabeçalho const hoje = new Date(); return dados.map((linha, index) => { let ultFreqData = linha[5] ? new Date(linha[5]) : null; let diasAusente = 0; let situacao = "Ativo"; let status = "Ativo"; // Recalcula dias ausentes e status automaticamente if (ultFreqData && !isNaN(ultFreqData.getTime())) { const diffTempo = hoje.getTime() - ultFreqData.getTime(); diasAusente = Math.max(0, Math.floor(diffTempo / (1000 * 3600 * 24))); if (diasAusente > 30 && diasAusente <= 60) { situacao = "Pré-evadido"; status = "Pré-evadido"; } else if (diasAusente > 60) { situacao = "Evadido"; status = "Evadido"; } } return { rowId: index + 2, // Posição exata da linha na planilha mat: linha[0] || "", aluno: linha[1] || "", turma: linha[2] || "Turma", nivel: linha[3] || "EM", contato: linha[4] || "", ultFreq: ultFreqData ? Utilities.formatDate(ultFreqData, "GMT-3", "yyyy-MM-dd") : "", situacao: situacao, status: status, diasAusente: diasAusente + " dias", professor: linha[9] || "Samuel", concludente: linha[10] === true || linha[10] === "SIM" }; }); } // 2. SALVAR AS ALTERAÇÕES REALIZADAS NA INTERFACE DE VOLTA NA PLANILHA function salvarAlteracoes(listaDados) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let aba = ss.getSheetByName('Matriculas') || ss.getSheets()[0]; listaDados.forEach(item => { if (item.rowId) { let range = aba.getRange(item.rowId, 1, 1, 11); range.setValues([[ item.mat, item.aluno, item.turma, item.nivel, item.contato, item.ultFreq, item.situacao, item.status, parseInt(item.diasAusente) || 0, item.professor, item.concludente ? "SIM" : "NÃO" ]]); } }); return { sucesso: true, mensagem: "Dados sincronizados com sucesso!" }; } catch (erro) { return { sucesso: false, mensagem: erro.message }; } } // 3. EXCLUIR REGISTRO DA PLANILHA function excluirAluno(rowId) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let aba = ss.getSheetByName('Matriculas') || ss.getSheets()[0]; aba.deleteRow(rowId); return obterDadosSincronizados(); } // 4. ZERAR BANCO DE DADOS (NOVO SEMESTRE) function zerarBancoDeDados() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let aba = ss.getSheetByName('Matriculas') || ss.getSheets()[0]; const ultimaLinha = aba.getLastRow(); if (ultimaLinha > 1) { aba.getRange(2, 1, ultimaLinha - 1, 11).clearContent(); } return []; }
